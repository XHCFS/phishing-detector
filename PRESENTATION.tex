\section{Risk Scoring: How We Assess Threats}

\begin{frame}{Intelligent Risk Scoring Algorithm}
\textbf{Every URL analyzed across 5 key factors:}

\vspace{0.3cm}
\begin{tabular}{llp{6cm}}
\toprule
\textbf{Factor} & \textbf{Weight} & \textbf{What It Checks} \\
\midrule
\textcolor{criticalred}{Liveness} & 0-35 pts & Is the malicious site currently active? \\[0.2cm]
\textcolor{highorange}{Recency} & 0-25 pts & Was it seen in the last few days? \\[0.2cm]
\textcolor{mediumyellow}{Domain Age} & 0-20 pts & Is it a newly created domain? \\[0.2cm]
\textcolor{codeblue}{TLD/Platform} & 0-10 pts & Suspicious TLD (.zip, .click)? \\[0.2cm]
\textcolor{lowgreen}{Keywords} & 0-10 pts & Contains "login", "verify", "bank"? \\
\bottomrule
\end{tabular}

\vspace{0.5cm}
\begin{block}{Risk Levels}

\textcolor{lowgreen}{\textbf{0-49}} & \textbf{Low} - Likely safe \\
\end{tabular}
\end{block}
\end{frame}
